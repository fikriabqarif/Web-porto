export default async function ProjectsPage() {
  const projects = await prisma.project.findMany({
    where: { published: true },
    orderBy: { createdAt: "desc" },
    include: { images: { orderBy: { sortOrder: "asc" }, take: 1 } },
  });

  return (
    <div className="mx-auto w-full max-w-5xl px-6">
      <header className="space-y-4 border-b border-border py-16">
        <p className="label">Index</p>
        <h1 className="text-3xl font-bold tracking-tight sm:text-5xl">Projects</h1>
        <p className="max-w-2xl text-sm leading-relaxed text-muted">
          Selected hardware and software work — measurement systems, embedded firmware and the boards around them.
        </p>
      </header>

      <section className="py-16">
        {projects.length === 0 ? (
          <p className="text-sm text-muted">No published projects yet.</p>
        ) : (
          <div className="grid gap-6 sm:grid-cols-2">
            {projects.map((project) => (
              <ProjectCard
                key={project.id}
                slug={project.slug}
                title={project.title}
                summary={project.summary}
                tags={project.tags}
                image={project.images[0]}
              />
            ))}
          </div>
        )}
      </section>
    </div>
  );
}

import { existsSync, readFileSync } from "node:fs";
import path from "node:path";
import { PrismaClient } from "@prisma/client";

// `tsx` does not read .env the way the Prisma CLI does, so load it here.
function loadEnv(): void {
  if (process.env.DATABASE_URL) return;
  const file = path.resolve(process.cwd(), ".env");
  if (!existsSync(file)) return;
  for (const line of readFileSync(file, "utf8").split("\n")) {
    const match = /^\s*([A-Z0-9_]+)\s*=\s*(.*)\s*$/.exec(line);
    if (!match) continue;
    const key = match[1] as string;
    const value = (match[2] ?? "").replace(/^["']|["']$/g, "");
    if (process.env[key] === undefined) process.env[key] = value;
  }
}

I was sampling faster than the source impedance allowed. With a 10 k potentiometer as the source, the sample-and-hold capacitor needs time to charge through that resistance. Halving the ADC clock prescaler and throwing away the first sample after switching channels took another 6 counts off. The rule of thumb: the source impedance and the sample capacitor form an RC network, and if you sample faster than five time constants you are measuring the previous channel.

## 3. Averaging, but the right kind

Ten readings, discard, ten more, then average. A plain moving average of 100 samples helped less than I expected because the noise was not white: it had a strong component at the mains frequency and its harmonics. An average over exactly one mains period (20 ms at 50 Hz) cancels those components almost completely. That is the trick that made the reading stop flickering on the display.

## 4. Decoupling close to the pin

A 100 nF capacitor from the ADC input to analog ground, plus a 100 nF on the reference pin, removed the last visible jitter. The capacitor has to be physically close; on a breadboard, close means the same row, not the same board.

Oversampling to 12 bits in software: the noise was not random, so it just gave more decimal places of the same wrong number. Shielding the wires: the noise was conducted, not radiated.
